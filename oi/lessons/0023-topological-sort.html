1. 拓扑排序：DAG 的线性化

拓扑排序（Topological Sort）：给定一个有向图，输出一个节点序列，使得每条边的起点都在终点之前。

拓扑排序存在的充要条件：图是 DAG（有向无环图）。如果有环，拓扑排序不存在。

直觉：可以想象成一堆课程的前置依赖——"计算机组成"需要先学"数字逻辑"，"操作系统"需要先学"计算机组成"。拓扑排序就是排出一个合理的选课顺序。
Kahn 算法（BFS 版）DFS 后序逆序
思路不断移除入度为 0 的节点DFS 后序遍历，最后逆序
数据结构队列 + 入度数组递归栈 + visited[] 三态
环检测结果集 size < V → 有环DFS 中遇到「正在访问」→ 有环
代码风格迭代，无递归风险递归，深度大时可能栈溢出

2. Kahn 算法（BFS 版）：入度驱动

// 拓扑排序标准模板（Kahn 算法）
vector<int> topologicalSort(int n, vector<vector<int>>& edges) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);

    // 建图 + 统计入度
    for (auto& e : edges) {
        int u = e[0], v = e[1];
        g[u].push_back(v);
        indeg[v]++;
    }

    // 初始：所有入度为 0 的节点入队
    queue<int> q;
    for (int i = 0; i < n; i++)
        if (indeg[i] == 0) q.push(i);

    // 不断移除队首，将其后继的入度 -1
    vector<int> res;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        res.push_back(u);
        for (int v : g[u]) {
            if (--indeg[v] == 0)
                q.push(v);
        }
    }

    // 结果长度 < n → 有环（拓扑排序不存在）
    return res.size() == n ? res : vector<int>();
}

✅ Kahn 算法的正确性证明（三段式）

① 不变式：队列中的节点都是当前「入度为 0」的节点——它们没有未完成的前驱。
② 初始成立：刚开始时，所有入度 0 的节点入队。
③ 循环保持：每次出队 u 加入结果，它所有的后继 v 的入度减 1。如果 v 的入度变为 0 → v 的所有前驱都已被处理 → v 可以安全加入结果。
⟹ 如果所有节点都进了结果 → 顺序合法。如果没进完 → 还有节点入度 > 0 → 这些节点构成环。

为什么用队列？Kahn 算法不需要队列——用任何容器（vector、set、stack）都行，因为拓扑序不唯一。用队列给出的是「字典序优先」的版本之一
如果要输出字典序最小的拓扑序 → 把 queue 换成 priority_queue (min-heap)。（LeetCode 269 · 火星词典 需要这个变种）

3. 环检测：有向图 vs 无向图

环检测是拓扑排序的孪生问题——有环 = 拓扑排序不存在。但有向图和无向图的环检测方法不同。

有向图环检测（三色标记法）

// DFS 三色标记：0=未访问, 1=正在访问, 2=已访问完
bool hasCycle(vector<vector<int>>& g) {
    int n = g.size();
    vector<int> state(n, 0);
    function<bool(int)> dfs = [&](int u) {
        state[u] = 1;  // 正在访问
        for (int v : g[u]) {
            if (state[v] == 1) return true;  // ⚡ 遇到「正在访问」= 环！
            if (state[v] == 0 && dfs(v))
                return true;
        }
        state[u] = 2;  // 访问完成
        return false;
    };
    for (int i = 0; i < n; i++)
        if (state[i] == 0 && dfs(i)) return true;
    return false;
}

无向图环检测

// 无向图：DFS 时记录 parent。遇到 visited 且不是 parent → 有环
bool hasCycleUndirected(vector<vector<int>>& g) {
    int n = g.size();
    vector<bool> vis(n, false);
    function<bool(int,int)> dfs = [&](int u, int parent) {
        vis[u] = true;
        for (int v : g[u]) {
            if (!vis[v]) { if (dfs(v, u)) return true; }
            else if (v != parent) return true;  // 遇到已访问且不是父节点 = 环
        }
        return false;
    };
    for (int i = 0; i < n; i++)
        if (!vis[i] && dfs(i, -1)) return true;
    return false;
}
有向图无向图
环检测方法三色标记法（DFS 遇到 state=1）parent 法（DFS 遇到 visited 且不是 parent）
Kahn 也能检测✅ 结果 size < V❌ Kahn 只适用于有向图
并查集（0024 内容）❌ 不适用✅ 最简便的方法

4. 题 1：课程表 I — 环检测

LeetCode 207 · 难度：🟡 中等

numCourses 门课，prerequisites[i] = [aᵢ, bᵢ] 表示学 aᵢ 前必须先学 bᵢ。判断是否可能完成所有课程（= 图是否有环）。

建模：每门课是一个节点，先修关系是一条边 bᵢ → aᵢ（先学 bᵢ 才能学 aᵢ）。有向图有环 → 不可能。Kahn 算法 和 DFS 三色标记都可以。 
// Kahn 算法判断是否有环
bool canFinish(int n, vector<vector<int>>& prerequisites) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);

    for (auto& p : prerequisites) {
        int a = p[0], b = p[1];
        g[b].push_back(a);   // b → a
        indeg[a]++;
    }

    queue<int> q;
    for (int i = 0; i < n; i++)
        if (indeg[i] == 0) q.push(i);

    int cnt = 0;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        cnt++;
        for (int v : g[u])
            if (--indeg[v] == 0) q.push(v);
    }
    return cnt == n;  // 全部入队 = 无环
}
DFS 三色版本（面试备选）：上面的数字如果是 n=2000，DFS 递归深度最多 2000（一条长依赖链），栈安全。如果面试官问「还能怎么做」，给出三色标记法的 DFS 版本加分。
一句口诀：课程表 I → K ahn 数个数，课程表 II → K ahn 存序列。

5. 题 2：课程表 II — 输出拓扑序

LeetCode 210 · 难度：🟡 中等

和课程表 I 完全一样，但不仅要判断「是否可能」，还要输出任意一个合理的选课顺序。

// 和课程表 I 几乎一样，多一个 res 存结果
vector<int> findOrder(int n, vector<vector<int>>& prerequisites) {
    vector<vector<int>> g(n);
    vector<int> indeg(n, 0);

    for (auto& p : prerequisites) {
        int a = p[0], b = p[1];
        g[b].push_back(a);
        indeg[a]++;
    }

    queue<int> q;
    for (int i = 0; i < n; i++)
        if (indeg[i] == 0) q.push(i);

    vector<int> res;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        res.push_back(u);     // 唯一区别：记录出队顺序
        for (int v : g[u])
            if (--indeg[v] == 0) q.push(v);
    }

    if (res.size() != n) return {};  // 有环
    return res;
}
题目变形：如果要求输出字典序最小的拓扑序怎么办？把 queue 换成 priority_queue<int, vector<int>, greater<int>>（小根堆）。每次选当前入度为 0 的节点中编号最小的。
这是 LeetCode 269 · 火星词典（Hard）的前置知识——它需要先建图再用 Kahn + 字典序最小。

6. 题 3：除法求值 — 带权图的 DFS/BFS

LeetCode 399 · 难度：🟡 中等

已知 a / b = 2.0, b / c = 3.0 … 求 a / c, a / e 等的值。

建图思路：每个变量是节点，除法关系是带权有向边。
a / b = 2.0 → 建两条边：a → b 权重 2.0（a = 2.0 × b），b → a 权重 0.5。
求 a / c = a 到 c 路径上权重的乘积：w(a→b) × w(b→c)。 
// DFS 带权图的路径乘积
using Node = pair<int, double>;  // (邻接节点, 权重)

vector<double> calcEquation(
    vector<vector<string>>& equations,
    vector<double>& values,
    vector<vector<string>>& queries)
{
    // 给变量名分配整数 id
    unordered_map<string, int> id;
    int n = 0;
    for (auto& e : equations) {
        if (!id.count(e[0])) id[e[0]] = n++;
        if (!id.count(e[1])) id[e[1]] = n++;
    }

    // 建带权无向图（双向边，权重互为倒数）
    vector<vector<Node>> g(n);
    for (int i = 0; i < equations.size(); i++) {
        int u = id[equations[i][0]], v = id[equations[i][1]];
        double w = values[i];
        g[u].push_back({v, w});
        g[v].push_back({u, 1.0 / w});
    }

    // DFS 求从 u 到 v 的路径权重乘积
    function<double(int,int,vector<bool>&)> dfs =
        [&](int u, int target, vector<bool>& vis) -> double {
        if (u == target) return 1.0;
        vis[u] = true;
        for (auto& [v, w] : g[u]) {
            if (!vis[v]) {
                double t = dfs(v, target, vis);
                if (t >= 0) return w * t;  // 找到路径
            }
        }
        return -1.0;  // 不可达
    };

    vector<double> ans;
    for (auto& q : queries) {
        if (!id.count(q[0]) || !id.count(q[1])) {
            ans.push_back(-1.0);
        } else {
            vector<bool> vis(n, false);
            ans.push_back(dfs(id[q[0]], id[q[1]], vis));
        }
    }
    return ans;
}
除法求值的三种解法（从易到难）：
① DFS / BFS（本课）— 每次查询单独遍历，O(N × (V+E))
② Floyd / 多源最短路 — 先算好所有对之间的值，O(V³)，但查询 O(1)
③ 并查集带权（0024 内容）— 合并时将权重关联到根，O(α(N)) 近似常数
面试中 DFS/BFS 足够，追问优化再提并查集。

7. 拓扑排序 vs DFS 后序遍历

DFS 后序逆序也能得到拓扑排序——很多资料只给这个版本。但它们的能力不同：

// DFS 后序逆序求拓扑排序（适合链长已知的场景）
void dfs(int u, vector<vector<int>>& g,
         vector<bool>& vis, vector<int>& res) {
    vis[u] = true;
    for (int v : g[u])
        if (!vis[v]) dfs(v, g, vis, res);
    res.push_back(u);  // 后序：子节点处理完后再加入
}
// 调用后 reverse(res.begin(), res.end()) → 拓扑序
Kahn（BFS）DFS 后序逆序
环检测✅ 结果长度 < V✅ 需要加三色标记
输出顺序按「前驱都处理完」顺序按「后继都处理完」逆序
偏好面试写这个——迭代无递归代码更短但递归有栈风险
面试建议：先写 Kahn 算法（迭代，入度驱动），然后在面试官追问「还能怎么做」时写出 DFS 后序逆序的版本。Kahn 的另一个优势是天然支持动态添加边——入度实时更新即可。
